cmake_minimum_required(VERSION 3.25)
project(sp01 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "SP-01 requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SCS_SDK_ROOT "" CACHE PATH "Extracted official SCS SDK 1.15 directory")
if(NOT EXISTS "${SCS_SDK_ROOT}/include/scssdk_telemetry.h")
  message(FATAL_ERROR "Set SCS_SDK_ROOT to the verified official SDK extraction")
endif()
add_library(sp01_transport STATIC native/transport.cpp)
target_include_directories(sp01_transport PUBLIC native)
target_compile_definitions(sp01_transport PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(sp01_transport PUBLIC advapi32 user32 ole32)
add_library(ets2telemetry_sp01 SHARED native/plugin.cpp native/plugin.def)
target_include_directories(ets2telemetry_sp01 PRIVATE "${SCS_SDK_ROOT}/include")
target_link_libraries(ets2telemetry_sp01 PRIVATE sp01_transport)
add_executable(sp01_native_test native/test.cpp)
target_link_libraries(sp01_native_test PRIVATE sp01_transport)
add_executable(sp01_sdk_test native/sdk-test.cpp native/plugin.cpp)
target_include_directories(sp01_sdk_test PRIVATE "${SCS_SDK_ROOT}/include")
target_compile_definitions(sp01_sdk_test PRIVATE SP01_SDK_TEST)
target_link_libraries(sp01_sdk_test PRIVATE sp01_transport)
foreach(target sp01_transport ets2telemetry_sp01 sp01_native_test sp01_sdk_test)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
endforeach()
enable_testing()
add_test(NAME sp01_native COMMAND sp01_native_test)
add_test(NAME sp01_sdk_lifecycle COMMAND sp01_sdk_test)
set_tests_properties(sp01_native sp01_sdk_lifecycle PROPERTIES TIMEOUT 12)

cmake_minimum_required(VERSION 4.3)
project(ETS2TelemetryPlaceholder LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The scaffold requires Windows x64 and MSVC.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ets2telemetry-placeholder SHARED src/placeholder.cpp)
target_compile_options(ets2telemetry-placeholder PRIVATE /W4 /WX)
include(CTest)
add_executable(placeholder-test tests/placeholder-test.cpp)
target_link_libraries(placeholder-test PRIVATE ets2telemetry-placeholder)
add_test(NAME placeholder-contract COMMAND placeholder-test)
set_tests_properties(placeholder-contract PROPERTIES TIMEOUT 60)
add_library(protocol-v0 STATIC src/protocol-v0.cpp)
target_include_directories(protocol-v0 PUBLIC src)
target_compile_options(protocol-v0 PRIVATE /W4 /WX)
add_executable(protocol-v0-test tests/protocol-v0-test.cpp)
target_link_libraries(protocol-v0-test PRIVATE protocol-v0)
target_compile_options(protocol-v0-test PRIVATE /W4 /WX)
add_test(NAME protocol-v0-contract COMMAND protocol-v0-test "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/fixtures/protocol-v0")
set_tests_properties(protocol-v0-contract PROPERTIES TIMEOUT 60)

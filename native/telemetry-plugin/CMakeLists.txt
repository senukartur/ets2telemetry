# Headers are pinned and shipped with the original permissive SDK notice.
set(SCS_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/scs-sdk-1.15")
file(STRINGS "${SCS_SDK_ROOT}/SHA256SUMS" SCS_SDK_HASHES)
foreach(SCS_SDK_ENTRY IN LISTS SCS_SDK_HASHES)
  string(SUBSTRING "${SCS_SDK_ENTRY}" 0 64 EXPECTED_HASH)
  string(SUBSTRING "${SCS_SDK_ENTRY}" 66 -1 SDK_FILE)
  file(SHA256 "${SCS_SDK_ROOT}/${SDK_FILE}" ACTUAL_HASH)
  if(NOT ACTUAL_HASH STREQUAL EXPECTED_HASH)
    message(FATAL_ERROR "Pinned SCS SDK file changed: ${SDK_FILE}")
  endif()
endforeach()

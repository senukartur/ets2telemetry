cmake_minimum_required(VERSION 3.24)
project(ets2telemetry_media_session_reader LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Media session reader requires Windows x64")
endif()
option(ETS2TELEMETRY_MEDIA_ASAN "Build media reader lifetime tests with AddressSanitizer" OFF)
set(ETS2TELEMETRY_CHANNEL "staging" CACHE STRING "Build channel")
if(NOT ETS2TELEMETRY_CHANNEL STREQUAL "staging" AND NOT ETS2TELEMETRY_CHANNEL STREQUAL "production")
  message(FATAL_ERROR "Invalid build channel")
endif()
function(media_target name source)
  add_executable(${name} ${source})
  target_compile_features(${name} PRIVATE cxx_std_17)
  target_compile_definitions(${name} PRIVATE _SILENCE_EXPERIMENTAL_COROUTINE_DEPRECATION_WARNINGS UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${name} PRIVATE /W4 /WX /EHsc /utf-8)
  target_link_libraries(${name} PRIVATE windowsapp)
endfunction()
media_target(media-session-reader src/main.cpp)
media_target(media-session-reader-fixture src/main.cpp)
target_compile_definitions(media-session-reader-fixture PRIVATE MEDIA_READER_FIXTURE=1)
enable_testing()
media_target(media-session-reader-tests tests/protocol.cpp)
add_test(NAME media-session-reader-protocol COMMAND media-session-reader-tests)
media_target(media-session-reader-policy-tests tests/policy.cpp)
foreach(family selection artwork lifetime)
  add_test(NAME media-session-reader-${family} COMMAND media-session-reader-policy-tests ${family})
endforeach()
if(ETS2TELEMETRY_MEDIA_ASAN)
  media_target(media-session-reader-asan-tests tests/protocol.cpp)
  target_compile_options(media-session-reader-asan-tests PRIVATE /fsanitize=address /Zi)
  target_link_options(media-session-reader-asan-tests PRIVATE /DEBUG /INCREMENTAL:NO)
  add_test(NAME media-session-reader-asan-protocol COMMAND media-session-reader-asan-tests)
  media_target(media-session-reader-asan-policy-tests tests/policy.cpp)
  target_compile_options(media-session-reader-asan-policy-tests PRIVATE /fsanitize=address /Zi)
  target_link_options(media-session-reader-asan-policy-tests PRIVATE /DEBUG /INCREMENTAL:NO)
  foreach(family selection artwork lifetime)
    add_test(NAME media-session-reader-asan-${family} COMMAND media-session-reader-asan-policy-tests ${family})
  endforeach()
  get_filename_component(msvc_binary_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
  foreach(target media-session-reader-asan-tests media-session-reader-asan-policy-tests)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${msvc_binary_directory}/clang_rt.asan_dynamic-x86_64.dll" "$<TARGET_FILE_DIR:${target}>")
  endforeach()
endif()
get_property(media_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${media_tests} PROPERTIES TIMEOUT 20)
